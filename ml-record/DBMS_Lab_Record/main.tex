\subsection*{Experiment 2: Database Schema Design}
\addcontentsline{toc}{subsection}{Experiment 2: Database Schema Design}
\textbf{Aim:}\\
Design a database schema for the above ER diagram. Clearly indicate tables, primary keys, and foreign keys.

\textbf{Query:}\\
\begin{lstlisting}
CREATE TABLE department (
    d_no INT PRIMARY KEY,
    d_name VARCHAR(50),
    location VARCHAR(100),
    no_of_emp INT
);

CREATE TABLE employee (
    emp_id SERIAL PRIMARY KEY,
    fname VARCHAR(50),
    mname VARCHAR(10),
    lname VARCHAR(50),
    sex CHAR(1),
    b_date DATE,
    address VARCHAR(100),
    salary NUMERIC(10,2),
    d_no INT REFERENCES department(d_no),
    experience_years INT
);

CREATE TABLE project (
    pno INT PRIMARY KEY,
    pname VARCHAR(100),
    location VARCHAR(100),
    d_no INT REFERENCES department(d_no)
);

CREATE TABLE works_on (
    hours INT,
    emp_id INT REFERENCES employee(emp_id),
    pno INT REFERENCES project(pno),
    PRIMARY KEY (emp_id, pno)
);

CREATE TABLE dependent (
    emp_id INT REFERENCES employee(emp_id),
    name VARCHAR(50),
    sex CHAR(1),
    b_date DATE,
    relationship VARCHAR(50),
    PRIMARY KEY (emp_id, name)
);
\end{lstlisting}

\textbf{(v) Create a package that uses a cursor to calculate and display the total working hours of each employee across all projects}\\

CREATE OR REPLACE PROCEDURE emp_pkg_total_hours() LANGUAGE plpgsql AS $$
DECLARE
    hours_cursor CURSOR FOR SELECT e.emp_id, e.fname, e.lname, COALESCE(SUM(w.hours), 0) AS total_hrs FROM employee e LEFT JOIN works_on w ON e.emp_id = w.emp_id GROUP BY e.emp_id, e.fname, e.lname;
    rec RECORD;
BEGIN
    RAISE NOTICE 'Total Working Hours Per Employee';
    OPEN hours_cursor;
    LOOP
        FETCH hours_cursor INTO rec;
        EXIT WHEN NOT FOUND;
        RAISE NOTICE 'ID: % | Name: % % | Total Hours: %', rec.emp_id, rec.fname, rec.lname, rec.total_hrs;
    END LOOP;
    CLOSE hours_cursor;
END;
$$;
